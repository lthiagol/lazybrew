// settle runs cmds to the end, as the program would: each message goes back
// through Update and the commands it returns run next. It does not follow the
// spinner's ticks, which run for as long as the program does. A command is
// waited for up to 120s: the status dashboard's own budget is 45s and each brew
// read's is 30s, so a shorter wait would give up before the app does. The
// preload's reads start from a landing, so a depth cut would drop them; the cut
// here is 64 levels, far past any chain.
func settle(m tea.Model, cmd tea.Cmd, depth int) tea.Model {
	if cmd == nil || depth > 64 {
		return m
	}
	ch := make(chan tea.Msg, 1)
	go func() { ch <- cmd() }()
	var msg tea.Msg
	select {
	case msg = <-ch:
	case <-time.After(120 * time.Second):
		return m
	}
	if msg == nil {
		return m
	}
	if _, ok := msg.(spinner.TickMsg); ok {
		return m
	}
	if b, ok := msg.(tea.BatchMsg); ok {
		for _, c := range b {
			m = settle(m, c, depth+1)
		}
		return m
	}
	if reflect.TypeOf(msg).String() == "tea.sequenceMsg" {
		v := reflect.ValueOf(msg)
		for i := 0; i < v.Len(); i++ {
			c := v.Index(i).Interface().(tea.Cmd)
			m = settle(m, c, depth+1)
		}
		return m
	}
	nm, nc := m.Update(msg)
	return settle(nm, nc, depth+1)
}

// drain runs cmds synchronously, skipping anything that takes > 3s (ticks).
func drain(m tea.Model, cmd tea.Cmd, depth int) tea.Model {
	if cmd == nil || depth > 6 {
		return m
	}
	ch := make(chan tea.Msg, 1)
	go func() { ch <- cmd() }()
	var msg tea.Msg
	select {
	case msg = <-ch:
	case <-time.After(20 * time.Second):
		return m
	}
	if msg == nil {
		return m
	}
	if b, ok := msg.(tea.BatchMsg); ok {
		for _, c := range b {
			m = drain(m, c, depth+1)
		}
		return m
	}
	if reflect.TypeOf(msg).String() == "tea.sequenceMsg" {
		v := reflect.ValueOf(msg)
		for i := 0; i < v.Len(); i++ {
			c := v.Index(i).Interface().(tea.Cmd)
			m = drain(m, c, depth+1)
		}
		return m
	}
	nm, nc := m.Update(msg)
	return drain(nm, nc, depth+1)
}

func key(m tea.Model, k tea.KeyPressMsg) tea.Model {
	nm, c := m.Update(k)
	return drain(nm, c, 0)
}

func runes(s string) tea.KeyPressMsg { return tea.KeyPressMsg{Code: []rune(s)[0], Text: s} }

func TestDump(t *testing.T) {
	// The Unicode icon set, so the frames read as plain text (AC-02.4).
	os.Setenv(config.IconsEnv, config.IconsUnicode)
	// Animations off (AC-06.1): every frame is the resting frame.
	os.Setenv(config.AnimationsEnv, "0")
	r, err := brew.NewDefaultRunner()
	if err != nil {
		t.Fatal(err)
	}
	// The background preload is on (gui.preload, the production default), set
	// here so the frames do not depend on the default. Every panel is read
	// before the first frame: settle runs the whole preload, and the check
	// below says so if a panel is still waiting.
	cfg := config.Default()
	cfg.GUI.Preload = true
	for _, sz := range [][2]int{{140, 40}, {80, 24}} {
		var m tea.Model = gui.New(brew.NewClient(r), cfg)
		m = settle(m, m.Init(), 0)
		nm, c := m.Update(tea.WindowSizeMsg{Width: sz[0], Height: sz[1]})
		m = settle(nm, c, 0)
		if n := strings.Count(ansi.Strip(m.View().Content), "not loaded"); n > 1 {
			t.Fatalf("%d sidebar boxes say not loaded after the preload; only Search should (it is never preloaded)", n)
		}
		out := fmt.Sprintf("%s/%dx%d", os.Getenv("DUMP_DIR"), sz[0], sz[1])
		os.MkdirAll(out, 0o755)
		shot := func(name string) {
			v := m.View().Content
			// The .ansi frames are kept for 140x40 only, as before.
			if sz[0] == 140 {
				os.WriteFile(out+"/"+name+".ansi", []byte(v), 0o644)
			}
			os.WriteFile(out+"/"+name+".txt", []byte(ansi.Strip(v)), 0o644)
		}
		shot("01-status")
		m = key(m, runes("2"))
		shot("02-formulae")
		m = key(m, runes("j"))
		shot("03-formulae-info")
		m = key(m, runes("]"))
		shot("04-formulae-deps")
		m = key(m, runes("4"))
		shot("05-outdated")
		m = key(m, runes("5"))
		shot("06-taps")
		m = key(m, runes("2"))
		m = key(m, runes("x"))
		shot("07-confirm-modal")
		m = key(m, tea.KeyPressMsg{Code: tea.KeyEsc})
		m = key(m, runes("?"))
		shot("08-help")
		m = key(m, tea.KeyPressMsg{Code: tea.KeyEsc})
		m = key(m, runes("7"))
		shot("09-search")
		// New in v2-02. Search takes focus when it opens, so Esc first.
		// The cursor on Formulae is still on bash (the second row, after
		// the j above), so Deps is bash's dependency list.
		m = key(m, tea.KeyPressMsg{Code: tea.KeyEsc})
		m = key(m, runes("2"))
		m = key(m, runes("]"))
		shot("11-deps-bash")
		// A toast over the Formulae list. TaskCompletedMsg sets the toast
		// synchronously; its dismissal tick is deliberately not drained, so
		// the toast is still on screen when the frame is taken.
		m = key(m, runes("2"))
		nm, _ = m.Update(gui.TaskCompletedMsg{ID: "harness-toast", Title: "Install bash"})
		m = nm
		shot("10-toast")
		// New in v2-04. The upgrade confirm opens on the bash row the cursor
		// is on; Esc cancels it. Nothing is run: 'u' only opens the dialog.
		m = key(m, runes("2"))
		m = key(m, runes("u"))
		shot("12-confirm-upgrade")
		m = key(m, tea.KeyPressMsg{Code: tea.KeyEsc})
		// The input modal for adding a tap. Esc cancels; no tap is added.
		m = key(m, runes("5"))
		m = key(m, runes("a"))
		shot("14-input-modal")
		m = key(m, tea.KeyPressMsg{Code: tea.KeyEsc})
		// Three toasts of different types, oldest first: success, error and
		// warning. The warning is "Cannot untap official taps", which needs an
		// official tap on the Taps list; this machine has none, so one is
		// injected as a list entry. Nothing runs: the untap is refused.
		nm, _ = m.Update(gui.TaskCompletedMsg{ID: "harness-ok", Title: "Install bash"})
		m = nm
		nm, _ = m.Update(gui.TaskCompletedMsg{ID: "harness-err", Title: "Upgrade bash", Err: errors.New("brew upgrade failed")})
		m = nm
		nm, _ = m.Update(gui.DataLoadedMsg{PanelID: gui.PanelTaps, Items: []string{"homebrew/cask-versions  official"}})
		m = nm
		m = key(m, runes("x"))
		shot("13-toast-stack")
		// New in v2-04 r3: the Brewfile menu, for the menu frame. Only opened
		// and closed: Enter would run a brew bundle action, so it is not sent.
		m = key(m, runes("1"))
		m = key(m, runes("B"))
		shot("15-menu-brewfile")
		m = key(m, tea.KeyPressMsg{Code: tea.KeyEsc})
		// New in v2-05. Help scrolls: one page down, so the first rows are
		// gone and the position hint reads from the second page. Esc closes it.
		m = key(m, runes("?"))
		m = key(m, tea.KeyPressMsg{Code: tea.KeyPgDown})
		shot("16-help-scrolled")
		m = key(m, tea.KeyPressMsg{Code: tea.KeyEsc})
		// The first cask's Info: Casks, whose Info tab is the default.
		m = key(m, runes("3"))
		shot("17-cask-info")
		// The Doctor tab: the doctor run the dashboard load already made is
		// cached, so this reads it and runs nothing new.
		m = key(m, runes("1"))
		m = key(m, runes("]"))
		m = key(m, runes("]"))
		shot("18-doctor")
		// The toasts of frames 10 and 13 are still up, and the error one lives
		// 6 s. Wait past that, then send the tick that drops the expired toasts
		// through the normal path, so frames 19 and 20 show the Services screen
		// and not those toasts over it (F-r1-03).
		time.Sleep(7 * time.Second)
		nm, c = m.Update(modal.ToastTickMsg{})
		m = drain(nm, c, 0)
		// New in v2-05s. Services, then j to the second row: moshi-hook, which
		// is running. The Info tab shows its detail and the key line the
		// actions that apply to a running service.
		m = key(m, runes("6"))
		m = key(m, runes("j"))
		shot("19-services-info")
		// S opens the stop confirm for moshi-hook. Esc cancels it. This harness
		// never sends y or enter on a services confirm, so no brew services
		// write runs on this machine.
		m = key(m, runes("S"))
		shot("20-services-stop-confirm")
		m = key(m, tea.KeyPressMsg{Code: tea.KeyEsc})
	}
	fmt.Println("ok")
}
